    57→/**
    58→ * The unified review input — built either from a fresh extraction
    59→ * (MandateDraftResponse) or re-hydrated from a stored DRAFT (detail GET).
    60→ */
    61→export interface ReviewInput {
    62→  mandateId: string;
    63→  correlationId: string | null;
    64→  classification: ExtractionClassification;
    65→  title: string;
    66→  instruction: string;
    67→  draft: NormalizedMandateDraft;
    68→  missingFields: MissingMandateField[];
    69→  clarificationQuestion: string | null;
    70→  warnings: string[];
    71→  meta: MandateDraftMeta | null;
    72→}
    73→
    74→/** Fresh extraction → review input. */
    75→export function reviewInputFromDraftResponse(
    76→  response: MandateDraftResponse
    77→): ReviewInput {
    78→  return {
    79→    mandateId: response.mandateId,
    80→    correlationId: response.correlationId,
    81→    classification: response.classification,
    82→    title: response.title,
    83→    instruction: response.instruction,
    84→    draft: response.draft,
    85→    missingFields: response.missingFields,
    86→    clarificationQuestion: response.clarificationQuestion,
    87→    warnings: response.warnings,
    88→    meta: response.meta,
    89→  };
    90→}
    91→
    92→/** Stored DRAFT (detail GET) → review input; null when it is not a reviewable draft. */
    93→export function reviewInputFromDetail(
    94→  detail: MandateDetailResponse
    95→): ReviewInput | null {
    96→  if (detail.status !== "DRAFT" || !detail.draft) {
    97→    return null;
    98→  }
    99→  return {
   100→    mandateId: detail.id,
   101→    correlationId:
   102→      // Re-link the audit chain to the original extraction correlation.
   103→      detail.audit.find((event) => event.eventType === "MANDATE_DRAFT_CREATED")
   104→        ?.correlationId ?? null,
   105→    classification:
   106→      detail.draft.missingFields.length > 0 ||
   107→      detail.draft.clarificationQuestion !== null
   108→        ? "AMBIGUOUS"
   109→        : "CLEAR",
   110→    title: detail.title,
   111→    instruction: detail.instruction,
   112→    draft: detail.draft.fields,
   113→    missingFields: detail.draft.missingFields,
   114→    clarificationQuestion: detail.draft.clarificationQuestion,
   115→    warnings: [],
   116→    meta: null,
   117→  };
   118→}

   129→  validityDays: number | null;
   130→  allowedCategories: string[] | null;
   131→  blockedCategories: string[] | null;
   132→  allowedMerchants: string[] | null;
   133→  blockedMerchants: string[] | null;
   134→  approvalMode: ApprovalMode;
   135→}

   137→/** Compact "what the interpretation DID determine" line for the amber alert. */
   138→function understoodSummary(draft: NormalizedMandateDraft): string {
   139→  const parts: string[] = [];
   140→  if (draft.allowedCategories && draft.allowedCategories.length > 0) {
   141→    parts.push(`Category: ${draft.allowedCategories.join(", ")}`);
   142→  }
   143→  if (draft.currency) {
   144→    parts.push(`Currency: ${draft.currency}`);
   145→  }
   146→  if (draft.maxTotal !== null && draft.currency) {
   147→    parts.push(`Maximum total: ${formatMoney(draft.maxTotal, draft.currency)}`);
   148→  }
   149→  return parts.length > 0
   150→    ? `We understood: ${parts.join(" · ")}.`
   151→    : "We could not determine the key terms of this mandate.";
   152→}
